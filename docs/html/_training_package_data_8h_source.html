297 int startingMoneyChange() const {
298 // First we parse the dice string to get the absolute value of the change and then we roll it. The sign of the change is determined by whether the original value was positive or negative.
299 std::string starting_money_modifier_dice = starting_money_modifier_dice_; // Make a copy of the string to modify for parsing
300
301 // Get whether this is an increase or decrease in starting money
302 bool neg = starting_money_modifier_dice.substr(0, 1) == "-";
303
304 if (neg) {
305 // Remove the negative sign for parsing
306 starting_money_modifier_dice = starting_money_modifier_dice.substr(1);
307 }
308
309 // Get the number of dice and the number of sides on the dice from the string. The format is expected to be [number of dice]d[number of sides]
310 std::vector<std::string> dice_parts = tokenise(starting_money_modifier_dice, "d");
311
312 int num_dice = 1;
313 int num_sides = 10;
314 int val = 0;
315 if (dice_parts.size() > 1) { // We have multiple dice so no open ended rolls
316 num_dice = std::stoi(dice_parts[0]);
317 num_sides = std::stoi(dice_parts[1]);
318 val = Dice(num_sides, 0, 0).roll(num_dice).result();
319 } else { // Single die with open ended rolls
320 num_sides = std::stoi(dice_parts[0]);
321 val = Dice(num_sides, 1, 0).roll(true).result();
322 }
323
324 return (neg ? val * -1 : val);
325 }
