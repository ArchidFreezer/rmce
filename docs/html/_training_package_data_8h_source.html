269 int startingMoneyChange() const {
270 // First we parse the dice string to get the absolute value of the change and then we roll it. The sign of the change is determined by whether the original value was positive or negative.
271 std::string starting_money_modifier_dice = starting_money_modifier_dice_; // Make a copy of the string to modify for parsing
272
273 // Get whether this is an increase or decrease in starting money
274 bool neg = starting_money_modifier_dice.substr(0, 1) == "-";
275
276 if (neg) {
277 // Remove the negative sign for parsing
278 starting_money_modifier_dice = starting_money_modifier_dice.substr(1);
279 }
280
281 // Get the number of dice and the number of sides on the dice from the string. The format is expected to be [number of dice]d[number of sides]
282 std::vector<std::string> dice_parts = tokenise(starting_money_modifier_dice, "d");
283
284 int num_dice = 1;
285 int num_sides = 10;
286 int val = 0;
287 if (dice_parts.size() > 1) { // We have multiple dice so no open ended rolls
288 num_dice = std::stoi(dice_parts[0]);
289 num_sides = std::stoi(dice_parts[1]);
290 val = Dice(num_sides, 0, 0).roll(num_dice).result();
291 } else { // Single die with open ended rolls
292 num_sides = std::stoi(dice_parts[0]);
293 val = Dice(num_sides, 1, 0).roll(true).result();
294 }
295
296 return (neg ? val * -1 : val);
297 }
