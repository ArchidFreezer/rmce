361 int startingMoneyChange() const {
362
363 if (starting_money_modifier_dice_.empty()) return 0; // No change if the string is empty
364
365 // First we parse the dice string to get the absolute value of the change and then we roll it. The sign of the change is determined by whether the original value was positive or negative.
366 std::string starting_money_modifier_dice = starting_money_modifier_dice_; // Make a copy of the string to modify for parsing
367
368 // Get whether this is an increase or decrease in starting money
369 bool neg = starting_money_modifier_dice.substr(0, 1) == "-";
370
371 if (neg) {
372 // Remove the negative sign for parsing
373 starting_money_modifier_dice = starting_money_modifier_dice.substr(1);
374 }
375
376 // Get the number of dice and the number of sides on the dice from the string. The format is expected to be [number of dice]d[number of sides]
377 std::vector<std::string> dice_parts = tokenise(starting_money_modifier_dice, "d");
378
379 int num_dice = 1;
380 int num_sides = 10;
381 int val = 0;
382 if (dice_parts.size() > 1) { // We have multiple dice so no open ended rolls
383 num_dice = std::stoi(dice_parts[0]);
384 num_sides = std::stoi(dice_parts[1]);
385 val = Dice(num_sides, 0, 0).roll(num_dice).result();
386 } else { // Single die with open ended rolls
387 num_sides = std::stoi(dice_parts[0]);
388 val = Dice(num_sides, 1, 0).roll(true).result();
389 }
390
391 return (neg ? val * -1 : val);
392 }
