352 int startingMoneyChange() const {
353
354 if (starting_money_modifier_dice_.empty()) return 0; // No change if the string is empty
355
356 // First we parse the dice string to get the absolute value of the change and then we roll it. The sign of the change is determined by whether the original value was positive or negative.
357 std::string starting_money_modifier_dice = starting_money_modifier_dice_; // Make a copy of the string to modify for parsing
358
359 // Get whether this is an increase or decrease in starting money
360 bool neg = starting_money_modifier_dice.substr(0, 1) == "-";
361
362 if (neg) {
363 // Remove the negative sign for parsing
364 starting_money_modifier_dice = starting_money_modifier_dice.substr(1);
365 }
366
367 // Get the number of dice and the number of sides on the dice from the string. The format is expected to be [number of dice]d[number of sides]
368 std::vector<std::string> dice_parts = tokenise(starting_money_modifier_dice, "d");
369
370 int num_dice = 1;
371 int num_sides = 10;
372 int val = 0;
373 if (dice_parts.size() > 1) { // We have multiple dice so no open ended rolls
374 num_dice = std::stoi(dice_parts[0]);
375 num_sides = std::stoi(dice_parts[1]);
376 val = Dice(num_sides, 0, 0).roll(num_dice).result();
377 } else { // Single die with open ended rolls
378 num_sides = std::stoi(dice_parts[0]);
379 val = Dice(num_sides, 1, 0).roll(true).result();
380 }
381
382 return (neg ? val * -1 : val);
383 }
