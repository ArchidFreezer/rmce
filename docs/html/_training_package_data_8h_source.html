326 int startingMoneyChange() const {
327
328 if (starting_money_modifier_dice_.empty()) return 0; // No change if the string is empty
329
330 // First we parse the dice string to get the absolute value of the change and then we roll it. The sign of the change is determined by whether the original value was positive or negative.
331 std::string starting_money_modifier_dice = starting_money_modifier_dice_; // Make a copy of the string to modify for parsing
332
333 // Get whether this is an increase or decrease in starting money
334 bool neg = starting_money_modifier_dice.substr(0, 1) == "-";
335
336 if (neg) {
337 // Remove the negative sign for parsing
338 starting_money_modifier_dice = starting_money_modifier_dice.substr(1);
339 }
340
341 // Get the number of dice and the number of sides on the dice from the string. The format is expected to be [number of dice]d[number of sides]
342 std::vector<std::string> dice_parts = tokenise(starting_money_modifier_dice, "d");
343
344 int num_dice = 1;
345 int num_sides = 10;
346 int val = 0;
347 if (dice_parts.size() > 1) { // We have multiple dice so no open ended rolls
348 num_dice = std::stoi(dice_parts[0]);
349 num_sides = std::stoi(dice_parts[1]);
350 val = Dice(num_sides, 0, 0).roll(num_dice).result();
351 } else { // Single die with open ended rolls
352 num_sides = std::stoi(dice_parts[0]);
353 val = Dice(num_sides, 1, 0).roll(true).result();
354 }
355
356 return (neg ? val * -1 : val);
357 }
