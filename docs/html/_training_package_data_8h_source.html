313 int startingMoneyChange() const {
314
315 if (starting_money_modifier_dice_.empty()) return 0; // No change if the string is empty
316
317 // First we parse the dice string to get the absolute value of the change and then we roll it. The sign of the change is determined by whether the original value was positive or negative.
318 std::string starting_money_modifier_dice = starting_money_modifier_dice_; // Make a copy of the string to modify for parsing
319
320 // Get whether this is an increase or decrease in starting money
321 bool neg = starting_money_modifier_dice.substr(0, 1) == "-";
322
323 if (neg) {
324 // Remove the negative sign for parsing
325 starting_money_modifier_dice = starting_money_modifier_dice.substr(1);
326 }
327
328 // Get the number of dice and the number of sides on the dice from the string. The format is expected to be [number of dice]d[number of sides]
329 std::vector<std::string> dice_parts = tokenise(starting_money_modifier_dice, "d");
330
331 int num_dice = 1;
332 int num_sides = 10;
333 int val = 0;
334 if (dice_parts.size() > 1) { // We have multiple dice so no open ended rolls
335 num_dice = std::stoi(dice_parts[0]);
336 num_sides = std::stoi(dice_parts[1]);
337 val = Dice(num_sides, 0, 0).roll(num_dice).result();
338 } else { // Single die with open ended rolls
339 num_sides = std::stoi(dice_parts[0]);
340 val = Dice(num_sides, 1, 0).roll(true).result();
341 }
342
343 return (neg ? val * -1 : val);
344 }
