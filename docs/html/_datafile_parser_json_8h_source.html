237inline void DatafileParserJson::saveData(const std::string& filename) {
238 if (filename.empty()) return;
239
240 // Main tree
241 pt::ptree tree;
242
243 // Tree of spell lists
244 pt::ptree data;
245
246 std::set<std::string> keys{};
247 factory().keys<T>(keys);
248
249 for (std::string key : keys) {
250 try {
251 pt::ptree datum;
252 populateDatum(key, datum);
253 data.push_back(std::make_pair("", datum));
254 } catch (const std::out_of_range& e) {
255 std::cout << "Error: " << e.what() << std::endl;
256 }
257 }
258
259 tree.add_child(rootNode(), data);
260
261 pt::write_json(filename, tree);
262}
