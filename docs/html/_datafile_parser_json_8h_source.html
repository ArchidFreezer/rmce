429inline void DatafileParserJson::saveData(const std::string& filename) {
430 if (filename.empty()) return;
431
432 // Main tree
433 pt::ptree tree;
434
435 // Tree of spell lists
436 pt::ptree data;
437
438 std::set<std::string> keys{};
439 factory().keys<GameRuleData>(keys);
440
441 for (std::string key : keys) {
442 try {
443 pt::ptree datum;
444 populateDatum(key, datum);
445 data.push_back(std::make_pair("", datum));
446 } catch (const std::out_of_range& e) {
447 std::cout << "Error: " << e.what() << std::endl;
448 }
449 }
450
451 tree.add_child(rootNode(), data);
452
453 pt::write_json(filename, tree);
454}
