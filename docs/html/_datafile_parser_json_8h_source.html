314inline void DatafileParserJson::saveData(const std::string& filename) {
315 if (filename.empty()) return;
316
317 // Main tree
318 pt::ptree tree;
319
320 // Tree of spell lists
321 pt::ptree data;
322
323 std::set<std::string> keys{};
324 factory().keys<T>(keys);
325
326 for (std::string key : keys) {
327 try {
328 pt::ptree datum;
329 populateDatum(key, datum);
330 data.push_back(std::make_pair("", datum));
331 } catch (const std::out_of_range& e) {
332 std::cout << "Error: " << e.what() << std::endl;
333 }
334 }
335
336 tree.add_child(rootNode(), data);
337
338 pt::write_json(filename, tree);
339}
