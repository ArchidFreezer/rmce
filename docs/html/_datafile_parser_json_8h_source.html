547template<GameRuleDataObject GameRuleData>
548inline void DatafileParserJson::saveData(const std::string& filename) {
549 if (filename.empty()) return;
550
551 // Main tree
552 pt::ptree tree;
553
554 // Tree of spell lists
555 pt::ptree data;
556
557 std::set<std::string> keys{};
558 factory().keys<GameRuleData>(keys);
559
560 for (std::string key : keys) {
561 try {
562 pt::ptree datum;
563 populateDatum(key, datum);
564 data.push_back(std::make_pair("", datum));
565 } catch (const std::out_of_range& e) {
566 std::cout << "Error: " << e.what() << std::endl;
567 }
568 }
569
570 tree.add_child(rootNode(), data);
571
572 pt::write_json(filename, tree);
573}
