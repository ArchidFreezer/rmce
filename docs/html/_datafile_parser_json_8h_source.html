465template<GameRuleDataObject GameRuleData>
466inline void DatafileParserJson::saveData(const std::string& filename) {
467 if (filename.empty()) return;
468
469 // Main tree
470 pt::ptree tree;
471
472 // Tree of spell lists
473 pt::ptree data;
474
475 std::set<std::string> keys{};
476 factory().keys<GameRuleData>(keys);
477
478 for (std::string key : keys) {
479 try {
480 pt::ptree datum;
481 populateDatum(key, datum);
482 data.push_back(std::make_pair("", datum));
483 } catch (const std::out_of_range& e) {
484 std::cout << "Error: " << e.what() << std::endl;
485 }
486 }
487
488 tree.add_child(rootNode(), data);
489
490 pt::write_json(filename, tree);
491}
