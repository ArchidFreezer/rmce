392template<class T>
393inline void DatafileParserJson::saveData(const std::string& filename) {
394 if (filename.empty()) return;
395
396 // Main tree
397 pt::ptree tree;
398
399 // Tree of spell lists
400 pt::ptree data;
401
402 std::set<std::string> keys{};
403 factory().keys<T>(keys);
404
405 for (std::string key : keys) {
406 try {
407 pt::ptree datum;
408 populateDatum(key, datum);
409 data.push_back(std::make_pair("", datum));
410 } catch (const std::out_of_range& e) {
411 std::cout << "Error: " << e.what() << std::endl;
412 }
413 }
414
415 tree.add_child(rootNode(), data);
416
417 pt::write_json(filename, tree);
418}
