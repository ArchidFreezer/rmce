200inline void DatafileParserJson::saveData(const std::string& filename) {
201 if (filename.empty()) return;
202
203 // Main tree
204 pt::ptree tree;
205
206 // Tree of spell lists
207 pt::ptree data;
208
209 std::set<std::string> keys{};
210 factory().keys<T>(keys);
211
212 for (std::string key : keys) {
213 try {
214 pt::ptree datum;
215 populateDatum(key, datum);
216 data.push_back(std::make_pair("", datum));
217 } catch (const std::out_of_range& e) {
218 std::cout << "Error: " << e.what() << std::endl;
219 }
220 }
221
222 tree.add_child(rootNode(), data);
223
224 pt::write_json(filename, tree);
225}
