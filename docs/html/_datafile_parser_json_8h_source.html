164inline void DatafileParserJson::saveData(const std::string& filename) {
165 if (filename.empty()) return;
166
167 // Main tree
168 pt::ptree tree;
169
170 // Tree of spell lists
171 pt::ptree data;
172
173 std::set<std::string> keys{};
174 factory().keys<T>(keys);
175
176 for (std::string key : keys) {
177 try {
178 pt::ptree datum;
179 populateDatum(key, datum);
180 data.push_back(std::make_pair("", datum));
181 } catch (const std::out_of_range& e) {
182 std::cout << "Error: " << e.what() << std::endl;
183 }
184 }
185
186 tree.add_child(rootNode(), data);
187
188 pt::write_json(filename, tree);
189}

224 std::map<SubcategoriedSkillData, U> datum{};
